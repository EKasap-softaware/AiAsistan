using Asistan.Application.Etiketler;
using Asistan.Domain.Entities;
using FluentAssertions;
using Moq;

namespace Asistan.Tests;

public class EtiketServisiTests
{
    private readonly Mock<IEtiketDeposu> _depo = new();
    private readonly EtiketServisi _servis;

    public EtiketServisiTests() => _servis = new EtiketServisi(_depo.Object);

    [Fact]
    public async Task OlusturAsync_GecerliAd_KirpipKaydeder()
    {
        var sonuc = await _servis.OlusturAsync("  is  ");

        sonuc.Ad.Should().Be("is");
        _depo.Verify(d => d.EkleAsync(It.Is<Etiket>(e => e.Ad == "is"), It.IsAny<CancellationToken>()), Times.Once);
    }

    [Theory]
    [InlineData("")]
    [InlineData("   ")]
    public async Task OlusturAsync_BosAd_HataVerirKaydetmez(string ad)
    {
        var islem = () => _servis.OlusturAsync(ad);

        await islem.Should().ThrowAsync<ArgumentException>();
        _depo.Verify(d => d.EkleAsync(It.IsAny<Etiket>(), It.IsAny<CancellationToken>()), Times.Never);
    }

    [Fact]
    public async Task OlusturAsync_51KarakterliAd_HataVerir()
    {
        var islem = () => _servis.OlusturAsync(new string('a', 51));

        await islem.Should().ThrowAsync<ArgumentException>();
    }
}
